# Génère podcast.xml (flux RSS du podcast) à partir de data/podcast.json.
# Usage : python3 podcast_rss.py <dossier_du_site>
import json, sys, os, datetime
from xml.sax.saxutils import escape as e
BASE = "https://le-sociologue-economie.pages.dev/"
root = sys.argv[1] if len(sys.argv) > 1 else "."
P = json.load(open(os.path.join(root, "data/podcast.json"), encoding="utf-8"))
def rfc(d): return datetime.datetime.strptime(d, "%Y-%m-%d").replace(hour=8, tzinfo=datetime.timezone.utc).strftime("%a, %d %b %Y %H:%M:%S +0000")
items = []
for ep in sorted(P["episodes"], key=lambda x: (x.get("date", ""), x["numero"]), reverse=True):
    if not ep.get("audio"): continue
    desc = ep["resume"] + "\n\nSources : " + " ; ".join(ep.get("sources", []))
    items.append(f"""  <item>
    <title>{e(ep['titre'])}</title>
    <link>{BASE}#podcast-{ep['id']}</link>
    <guid isPermaLink="false">{ep['id']}</guid>
    <pubDate>{rfc(ep['date'])}</pubDate>
    <description>{e(desc)}</description>
    <enclosure url="{BASE}{ep['audio']}" length="{ep.get('taille', 0)}" type="audio/mpeg"/>
    <itunes:duration>{ep.get('duree_s', 0)}</itunes:duration>
    <itunes:episode>{ep['numero']}</itunes:episode>
  </item>""")
xml = f"""<?xml version="1.0" encoding="UTF-8"?>
<rss version="2.0" xmlns:itunes="http://www.itunes.com/dtds/podcast-1.0.dtd">
<channel>
  <title>{e(P['titre'])}</title>
  <link>{BASE}#podcast</link>
  <language>fr</language>
  <description>{e(P['presentation'] + ' ' + P['voix'])}</description>
  <itunes:author>Le Sociologue de l'Économie (blog de Dr Fabrice TANOH)</itunes:author>
  <itunes:image href="{BASE}img/podcast-cover.jpg"/>
  <itunes:category text="Science"><itunes:category text="Social Sciences"/></itunes:category>
  <itunes:explicit>false</itunes:explicit>
{chr(10).join(items)}
</channel>
</rss>
"""
open(os.path.join(root, "podcast.xml"), "w", encoding="utf-8").write(xml)
print("podcast.xml :", len(items), "épisodes")
